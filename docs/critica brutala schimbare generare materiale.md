>
> Tehnica se numește '5 De Ce-uri' și vine de la Toyota. Ideea e simplă: când ai o problemă, nu te opri la primul răspuns. Sapă mai adânc. De obicei, cauza reală e la al 3-lea sau 4-lea 'De ce?'.

### DEBRIEF COLECTIV (3 minute - CRITICE)

**Obiectiv:** Extrage învățăminte, nu doar să auzi rapoarte.

**Script facilitator (întrebări în ordine):**

1. **"Cine a descoperit că cauza rădăcină era STRUCTURALĂ, nu personală?"**
   - Lasă 3-4 oameni să ridice mâna și să dea exemple de 20 secunde fiecare
   - Validează: "Perfect. Exact asta căutam. Procesul, nu persoana."

2. **"Și acum întrebarea de un milion: Cine poate SCHIMBA cauza pe care ați găsit-o?"**
   - Dacă zic "Nu pot, e decizie de la nivel de management":
     → "Ok, atunci mergeți la următorul 'De ce' până găsiți ceva în controlul vostru."
   - Dacă zic "Eu pot":
     → "Excelent. Aveți 30 de secunde să scrieți în workbook CE veți face LUNI. Specific, nu vag."

3. **"Ultimă întrebare: Ce diferență e între ce ați făcut acum și ce faceți de obicei când aveți un conflict?"**
   - Lasă-i să vadă singuri insight-ul: "De obicei atacăm persoana, acum am atacat procesul."

**Tranziție la următoarea secțiune:**
> "Perfect. Țineți minte: Conflictul nu e despre CINE are dreptate. E despre CE proces e defect. Acum că știți asta, în următoarele 15 minute o să vedem cum să REPARAȚI procesul folosind modelul SBI. Stay with me."

typescriptfunction calculateRealisticDuration(
  activities: Activity[], 
  environment: 'LiveWorkshop' | 'OnlineCourse'
): TimingReport {
  
  const BUFFERS = {
    LiveWorkshop: {
      lecture: 1.2,        // +20% for Q&A
      individual: 1.1,     // +10% for slower writers
      group: 1.4,          // +40% for sync delays
      debrief: 1.5,        // +50% for extended discussions
      transition: 3,       // +3 min between activities
      break_after_90min: 15 // Mandatory 15 min break every 90 min
    },
    OnlineCourse: {
      lecture: 1.1,        // +10% for tech issues
      individual: 1.2,     // +20% for digital friction
      group: 1.3,          // +30% for breakout room delays
      debrief: 1.4,        // +40% for chat/unmute delays
      transition: 2,       // +2 min between videos
      break_after_60min: 10 // Recommend 10 min break every 60 min (online fatigue)
    }
  };
  
  const buffer = BUFFERS[environment];
  let totalDeclared = 0;
  let totalReal = 0;
  let warnings = [];
  
  for (const activity of activities) {
    const declared = activity.duration;
    const multiplier = buffer[activity.type];
    const real = declared * multiplier;
    
    totalDeclared += declared;
    totalReal += real;
    
    // Add transition time
    totalReal += buffer.transition;
  }
  
  // Add mandatory breaks
  const breakInterval = environment === 'LiveWorkshop' ? 90 : 60;
  const breaksNeeded = Math.floor(totalReal / breakInterval);
  const breakDuration = environment === 'LiveWorkshop' ? 15 : 10;
  totalReal += breaksNeeded * breakDuration;
  
  // Validate against blueprint
  const blueprintDuration = parseFloat(course.blueprint.estimated_duration);
  const overrun = totalReal - blueprintDuration;
  
  if (overrun > 0) {
    warnings.push(`⚠️ TIMING OVERRUN: Declared ${totalDeclared}min, Real ${totalReal}min, Blueprint ${blueprintDuration}min. OVER by ${overrun}min (${Math.round(overrun/blueprintDuration*100)}%)`);
    warnings.push(`💡 ACTION: Cut ${overrun}min of content OR increase blueprint duration.`);
  }
  
  return {
    totalDeclared,
    totalReal,
    overrun,
    warnings,
    breakdown: activities.map(a => ({
      activity: a.title,
      declared: a.duration,
      real: a.duration * buffer[a.type],
      buffer: `${Math.round((buffer[a.type]-1)*100)}%`
    }))
  };
}

typescriptinterface MasterIndex {

  exercises: Map<string, ExerciseEntry>;

interface ConceptEntry {
  id: string;
  name: string;
  aliases: string[]; // ["SBI", "Situation-Behavior-Impact", "feedback model"]
  
  introduced_in: {
    type: 'slide' | 'lecture' | 'exercise';
    module: number;
    reference: string; // "Slide 8" or "Lecture 2.3"
  };

    type: 'slide' | 'workbook' | 'manual' | 'exercise';
    module: number;
    reference: string;
    context: string; // Why it's referenced
  }>;
  
  dependencies: string[]; // IDs of concepts that must be taught first
}

// Build index BEFORE generating documents
async function buildMasterIndex(blueprint: CourseBlueprint): Promise<MasterIndex> {
  const index: MasterIndex = {
    concepts: new Map(),
    exercises: new Map(),
    caseStudies: new Map()
  };
  
  // Extract concepts from learning objectives
  for (const module of blueprint.modules) {
    const concepts = extractConceptsFromObjective(module.learning_objective);
    
    for (const concept of concepts) {
      index.concepts.set(concept.id, {
        id: concept.id,
        name: concept.name,
        aliases: generateAliases(concept.name),
        introduced_in: { type: 'lecture', module: module.id, reference: `Module ${module.id}` },
        referenced_in: [],
        dependencies: []
      });
    }
  }
  
  return index;
}

// VALIDATE references during generation
function validateReference(
  reference: string, 
  currentModule: number, 
  index: MasterIndex
): ValidationResult {
  
  // Parse reference (e.g., "see Slide 8" or "using SBAR from Module 2")
  const parsed = parseReference(reference);
  
  if (parsed.type === 'concept') {
    const concept = index.concepts.get(parsed.id);
    
    if (!concept) {
      return { valid: false, error: `Referenced concept "${parsed.name}" doesn't exist in index` };
    }
    
    if (concept.introduced_in.module > currentModule) {
      return { 
        valid: false, 
        error: `Forward reference: Module ${currentModule} references "${concept.name}" but it's introduced in Module ${concept.introduced_in.module}` 
      };
    }
  }
  
  if (parsed.type === 'slide') {
    // Validate slide number exists
    const slideExists = checkSlideExists(parsed.module, parsed.slideNumber);
    if (!slideExists) {
      return { valid: false, error: `Slide ${parsed.slideNumber} doesn't exist in Module ${parsed.module}` };
    }
  }
  
  return { valid: true };
}
